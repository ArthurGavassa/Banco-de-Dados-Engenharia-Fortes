--PRIMEIRO SELECT: Quantidade de horas extras aprovadas para cada funcionário.

SELECT
    t.cod,
    t.nome AS tecnicos,
    COUNT(*) AS qtd_horas_extras_aprovadas
FROM tecnico t
JOIN info_horas_extras_tec ih 
    ON ih.cod_tecnico = t.cod
WHERE 
    ih.status = 'A'
GROUP BY 
    t.cod,
    t.nome
ORDER BY 
    qtd_horas_extras_aprovadas DESC;


--SEGUNDO SELECT: Quantas solicitações de cada encarregado foram aprovadas.

SELECT 
    e.nome AS encarregado,
    COUNT(hg.*) AS qtd_solicitacoes_aprovadas
FROM encarregado e
LEFT JOIN horas_gestor hg
    ON hg.encarregado = e.nome        
   AND hg.status = 'A'                
   AND hg.tipo_registro = 'S'        
GROUP BY 
    e.nome
ORDER BY 
    qtd_solicitacoes_aprovadas DESC;
