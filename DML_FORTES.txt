-- BLOCO 5: INFO_TECNICOS (400 linhas)
-- ==========================================
WITH base AS (
  SELECT 
    g AS cod,
    ((g-1) % 40) + 1      AS cod_tec,
    80 + ((g-1) % 121)    AS total_horas_mes,
    (20 + ((g-1) % 16))::numeric(10,2) AS valor_hora
  FROM generate_series(1, 400) g
)
INSERT INTO info_tecnicos (cod, nome, total_horas_mes, valor_total_hora_extra, cod_encarregado, cod_gestor)
SELECT
  b.cod,
  t.nome,
  b.total_horas_mes,
  (b.total_horas_mes * b.valor_hora)::numeric(10,2) AS valor_total_hora_extra,
  ((b.cod-1) % 12) + 1 AS cod_encarregado,
  ((b.cod-1) % 6)  + 1 AS cod_gestor
FROM base b
JOIN tecnico t ON t.cod = b.cod_tec;
