CREATE TABLE tecnico (
	cod INT,
    	nome VARCHAR,
    	email VARCHAR,
    	senha VARCHAR,
    	telefone VARCHAR,
    	CONSTRAINT pk_tecnico PRIMARY KEY(cod)
);

CREATE TABLE encarregado (
    	cod INT,
    	nome VARCHAR,
    	email VARCHAR,
    	senha VARCHAR,
    	telefone VARCHAR,
    	CONSTRAINT pk_encarregado PRIMARY KEY(cod)
);

CREATE TABLE gestor (
    	cod INT,
    	nome VARCHAR,
    	email VARCHAR,
   	senha VARCHAR,
   	telefone VARCHAR,
    	CONSTRAINT pk_gestor PRIMARY KEY(cod)
);

CREATE TABLE info_horas_extras_tec (
  	cod INT,
        nome VARCHAR,
	obra VARCHAR,
  	comentarios TEXT,
  	valor DECIMAL(10,2),
  	duracao_intervalo VARCHAR,
  	encarregado VARCHAR,
  	data_marcada DATE,
  	status CHAR,
  	cod_tecnico INT,
  	CONSTRAINT pk_info_horas_extras_tec PRIMARY KEY(cod),
  	CONSTRAINT fk_tecnico FOREIGN KEY(cod_tecnico) REFERENCES tecnico(cod)
);

CREATE TABLE info_tecnicos (
  	cod INT,
	nome VARCHAR,
  	total_horas_mes INT,
  	valor_total_hora_extra DECIMAL(10,2),
  	cod_encarregado INT,
  	cod_gestor INT,
  	CONSTRAINT pk_info_tecnicos PRIMARY KEY(cod),
  	CONSTRAINT fk_encarregado FOREIGN KEY(cod_encarregado) REFERENCES encarregado(cod),
  	CONSTRAINT fk_gestor FOREIGN KEY(cod_gestor) REFERENCES gestor(cod)
);

CREATE TABLE tab_status (
  	cod INT,
	data_envio DATE,
  	duracao_intervalo VARCHAR,
  	tecnico VARCHAR,
  	obra VARCHAR,
  	status CHAR,
  	total_aprovados INT,
  	total_em_analise INT,
  	total_recusados INT,
  	cod_encarregado INT,
  	CONSTRAINT pk_tab_status PRIMARY KEY(cod),

);
